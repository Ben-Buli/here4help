# CPanel .htaccess 修復配置

# 修復 Flutter Web 路徑重導問題
# 確保 /frontend/ 請求不會重導到內部檔案系統路徑

# 1. Flutter Web 靜態檔案優先處理
RewriteCond %{REQUEST_URI} ^/frontend/
RewriteCond %{REQUEST_FILENAME} -f
RewriteRule ^(.*)$ - [L]

# 2. Flutter Web SPA 路由
RewriteCond %{REQUEST_URI} ^/frontend/
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule ^frontend/(.*)$ frontend/index.html [L]

# 3. 根路徑重導到 Flutter Web (避免重導到內部路徑)
RewriteCond %{REQUEST_URI} ^/$
RewriteRule ^$ frontend/ [L,R=301]

# 4. 其他請求重導到 Flutter Web
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_URI} !^/backend
RewriteCond %{REQUEST_URI} !^/admin
RewriteCond %{REQUEST_URI} !^/api
RewriteCond %{REQUEST_URI} !^/socket
RewriteCond %{REQUEST_URI} !^/frontend
RewriteRule ^(.*)$ frontend/index.html [L]
